Problem 1: Factorial

def factorial(n):
    result = 1
    for i in range(1, n + 1):
        result *= i
    return result

# Used a loop because I find it easier to read than recursion.

Problem 2: Palindrome

def is_palindrome(s):
    s = s.lower()
    return s == s[::-1]

print(is_palindrome("Madam"))   # True
print(is_palindrome("Python"))  # False
