Problem 1: Factorial

def factorial(n):
    # Base case: factorial of 0 or 1 is 1.
    # Without this, the recursion would never stop since n would
    # decrease forever below 0 with no exit condition.
    if n == 0 or n == 1:
        return 1
    # Recursive case: n! = n * (n-1)!
    # Each call reduces n by 1, guaranteed to reach the base case.
    return n * factorial(n - 1)

# Test cases
print(factorial(0))   # 1
print(factorial(1))   # 1
print(factorial(5))   # 120
print(factorial(10))  # 3628800

Problem 2: Palindrome

def is_palindrome(s):
    # Normalize to lowercase so comparison is case-insensitive
    s = s.lower()
    # Reverse the string and compare directly
    return s == s[::-1]

# Test cases
print(is_palindrome("Racecar"))   # True
print(is_palindrome("Hello"))     # False
print(is_palindrome("Madam"))     # True
print(is_palindrome(""))          # True (empty string is a palindrome)
